# UI/UX Audit Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a monorepo-resident UI/UX audit tool (`tools/ui-audit/`) that runs end-to-end audits on `apps/web`, diffs against git-tracked baselines, posts PR comments, and integrates with GitHub Actions.

**Architecture:** Single Node package with one CLI entry (`audit.mjs`) and 11 focused internal modules. One code path serves both local dev and CI. Test-only auth is a Next.js route gated by `NODE_ENV !== 'production'`. Baselines are JSON files committed to git under `tools/ui-audit/baselines/`.

**Tech Stack:** Node.js (ESM, no TypeScript in the tool itself to keep it dependency-light), Playwright (`@playwright/test` for the browser API), `@axe-core/playwright`, `lighthouse` CLI, `wait-on`, `yaml` (for config parsing), `octokit` (for the PR comment API). Next.js 16 route handler for the test auth endpoint.

**Spec:** `docs/superpowers/specs/2026-09-05-ui-audit-design.md`

## Global Constraints

- Node ≥ 20, ESM-only (`"type": "module"` in every `package.json` this plan creates).
- pnpm workspace: `tools/*` must be added to `pnpm-workspace.yaml` (Task 1).
- Windows-compatible: no PowerShell file writes; no shell `&&` chains; all file I/O via `fs.writeFile` with explicit `utf8` encoding; `child_process.spawn` with explicit args.
- All written JSON files must be BOM-free (AGENTS.md Windows gotcha).
- Never write `.env` files. All test credentials read from `process.env`.
- Routes audited: `['/overview', '/analytics', '/behavioral', '/business', '/logs', '/reward']` — hardcoded in `lib/routes.mjs`.
- Viewports: `1440x900`, `768x1024`, `375x812`.
- Failure thresholds (per spec §5): P0 a11y, any console error, any broken asset, Lighthouse perf < 50 or regression > 10 → exit 1.
- The tool must never be invoked with `NODE_ENV=production`. The test auth endpoint self-enforces this; the audit CLI refuses to run if it is.

```javascript
export function compute({ findings = [], deltas = [], lighthouse = {} }) {
  const critical = [];
  const warn = [];

  for (const f of findings) {
    if (f.kind === 'a11y' && (f.impact === 'critical' || f.impact === 'serious')) {
      critical.push(f);
    } else if (f.kind === 'a11y' && (f.impact === 'moderate' || f.impact === 'minor')) {
      warn.push(f);
    } else if (f.kind === 'console' && f.severity === 'error') {
      critical.push(f);
    } else if (f.kind === 'console' && f.severity === 'warning') {
      warn.push(f);
    } else if (f.kind === 'pageerror') {
      critical.push(f);
    } else if (f.kind === 'broken-asset') {
      critical.push(f);
    } else if (f.kind === 'auth') {
      critical.push(f);
    } else if (f.kind === 'route-timeout') {
      critical.push(f);
    } else if (f.kind === 'selfcheck') {
      critical.push(f);
    }
  }

  if (typeof lighthouse.performance === 'number' && lighthouse.performance < 50) {
    critical.push({ kind: 'lighthouse-perf', route: lighthouse.route ?? '?', message: `Performance ${lighthouse.performance} < 50` });
  }
  if (typeof lighthouse.accessibility === 'number' && lighthouse.accessibility < 90) {
    critical.push({ kind: 'lighthouse-a11y', route: lighthouse.route ?? '?', message: `Accessibility ${lighthouse.accessibility} < 90` });
  }

  for (const d of deltas) {
    if (d.metric === 'performance' && d.delta <= -10) {
      critical.push({ kind: 'lighthouse-regression', route: d.route, message: `Performance regressed ${d.delta} points` });
    } else if (d.metric === 'performance' && d.delta <= -5) {
      warn.push({ kind: 'lighthouse-regression', route: d.route, message: `Performance regressed ${d.delta} points` });
    }
  }

  let grade;
  if (critical.length === 0 && warn.length === 0) grade = 'A';
  else if (critical.length === 0 && warn.length <= 2) grade = 'B';
  else if (critical.length === 0) grade = 'C';
  else if (critical.length <= 2) grade = 'D';
  else grade = 'F';

  return { grade, critical, warn };
}
```

```javascript
export async function withBackoff({ fn, attempts = 3, baseMs = 500, label = 'op' }) {
  let lastErr;
  for (let i = 0; i < attempts; i++) {
    try {
      return await fn();
    } catch (err) {
      lastErr = err;
      if (i === attempts - 1) break;
      const delay = baseMs * Math.pow(2, i);
      await new Promise((res) => setTimeout(res, delay));
    }
  }
  const msg = lastErr?.message ?? String(lastErr);
  const wrapped = new Error(`${label}: ${msg}`);
  wrapped.cause = lastErr;
  throw wrapped;
}
```

```javascript
import { spawnSync } from 'node:child_process';

export function detect() {
  if (process.env.GITHUB_ACTIONS !== 'true') return null;
  if (!process.env.GITHUB_TOKEN) return null;
  if (!process.env.GITHUB_REPOSITORY) return null;

  const [owner, repo] = process.env.GITHUB_REPOSITORY.split('/');
  const refMatch = (process.env.GITHUB_REF ?? '').match(/refs\/pull\/(\d+)\//);
  const pr = refMatch ? Number(refMatch[1]) : null;

  return {
    provider: 'github',
    owner,
    repo,
    pr,
    sha: process.env.GITHUB_SHA ?? null,
    baseRef: process.env.GITHUB_BASE_REF ?? 'main',
    baseSha: null,
    token: process.env.GITHUB_TOKEN,
    runId: process.env.GITHUB_RUN_ID ?? null,
  };
}

export function smokeLighthouse() {
  const r = spawnSync('npx', ['--no-install', 'lighthouse', '--version'], {
    encoding: 'utf8',
    shell: false,
  });
  if (r.status !== 0) {
    throw new Error(`lighthouse CLI not available: ${r.stderr || r.stdout}`);
  }
}
```

```javascript
import { spawn } from 'node:child_process';
import waitOn from 'wait-on';

export async function boot({ port = 3000, cwd = process.cwd(), command, env = {} }) {
  const child = spawn(command[0], command.slice(1), {
    cwd,
    stdio: ['ignore', 'pipe', 'pipe'],
    shell: false,
    env: { ...process.env, ...env },
    detached: true,
  });

  let exited = false;
  let exitInfo = null;
  child.on('exit', (code, signal) => {
    exited = true;
    exitInfo = { code, signal };
  });

  const url = `http://localhost:${port}`;

  try {
    await waitOn({
      resources: [url],
      timeout: 60_000,
      interval: 500,
      validateStatus: (s) => s >= 200 && s < 500,
    });
  } catch (err) {
    await stop();
    throw new Error(`Dev server failed to become ready on ${url}: ${err.message}`);
  }

  if (exited) {
    throw new Error(`Dev server exited before becoming ready (${JSON.stringify(exitInfo)})`);
  }

  async function stop() {
    if (exited) return;
    try {
      if (process.platform === 'win32') {
        spawn('taskkill', ['/pid', String(child.pid), '/T', '/F'], { shell: false });
      } else {
        process.kill(-child.pid, 'SIGKILL');
      }
    } catch {}
    await new Promise((res) => {
      if (exited) return res();
      child.once('exit', res);
      setTimeout(res, 2000);
    });
  }

  return { url, stop, child };
}
```

```javascript
import fs from 'node:fs';
import path from 'node:path';
import { spawnSync } from 'node:child_process';

const BASELINES_DIR = path.join('tools', 'ui-audit', 'baselines');

export function read(ref) {
  const p = path.join(BASELINES_DIR, `${ref}.json`);
  if (!fs.existsSync(p)) return null;
  const text = fs.readFileSync(p, 'utf8').replace(/^\uFEFF/, '');
  try {
    return JSON.parse(text);
  } catch (err) {
    throw new Error(`baseline ${ref} is not valid JSON: ${err.message}`);
  }
}

export function write(ref, data) {
  fs.mkdirSync(BASELINES_DIR, { recursive: true });
  const p = path.join(BASELINES_DIR, `${ref}.json`);
  fs.writeFileSync(p, JSON.stringify(data, null, 2) + '\n', { encoding: 'utf8' });
  return p;
}

export function resolveRef({ baseRef = 'main' } = {}) {
  const r = spawnSync('git', ['rev-parse', `origin/${baseRef}`], { encoding: 'utf8', shell: false });
  if (r.status === 0) return { ref: r.stdout.trim(), sha: r.stdout.trim() };
  return { ref: 'main', sha: null };
}

export function diff(current, baseline) {
  if (!baseline || !baseline.routes) return [];
  const deltas = [];
  const metrics = ['perf', 'a11yViolations', 'consoleErrors', 'lighthouseA11y'];
  for (const [route, cur] of Object.entries(current.routes ?? {})) {
    const base = baseline.routes[route];
    if (!base) continue;
    for (const m of metrics) {
      if (typeof cur[m] === 'number' && typeof base[m] === 'number' && cur[m] !== base[m]) {
        deltas.push({ route, metric: m, delta: cur[m] - base[m], before: base[m], after: cur[m] });
      }
    }
  }
  return deltas;
}
```

```javascript
import { chromium } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';
import fs from 'node:fs';
import path from 'node:path';

async function withTimeout(promise, ms, label) {
  let to;
  const timeout = new Promise((_, reject) => {
    to = setTimeout(() => reject(new Error(`${label} timed out after ${ms}ms`)), ms);
  });
  try {
    return await Promise.race([promise, timeout]);
  } finally {
    clearTimeout(to);
  }
}

export async function runRoute({ browser, baseUrl, route, viewports, cookieHeader, timeoutMs = 45_000 }) {
  const context = await browser.newContext({ reducedMotion: 'reduce' });
  if (cookieHeader) {
    const [name, ...rest] = cookieHeader.split('=');
    await context.addCookies([{ name, value: rest.join('='), url: baseUrl }]);
  }
  const findings = [];
  const perViewport = [];

  for (const vp of viewports) {
    const page = await context.newPage();
    const consoleErrors = [];
    const consoleWarns = [];
    const pageErrors = [];

    page.on('console', (msg) => {
      if (msg.type() === 'error') consoleErrors.push(msg.text());
      if (msg.type() === 'warning') consoleWarns.push(msg.text());
    });
    page.on('pageerror', (err) => pageErrors.push(err.message));

    let pvFindings = [];

    try {
      await withTimeout(
        (async () => {
          await page.setViewportSize({ width: vp.width, height: vp.height });
          await page.goto(`${baseUrl}${route}`, { waitUntil: 'networkidle' });
          await page.waitForFunction(() => document.readyState === 'complete');
          await page.waitForTimeout(1500);

          const axe = await new AxeBuilder({ page }).analyze();
          for (const v of axe.violations) {
            pvFindings.push({
              kind: 'a11y',
              impact: v.impact ?? 'minor',
              route,
              viewport: vp.name,
              message: `${v.id}: ${v.help} (${v.nodes.length} node${v.nodes.length === 1 ? '' : 's'})`,
            });
          }

          const imgs = await page.$$eval('img', (els) => els.map((e) => ({ src: e.src, alt: e.alt, nw: e.naturalWidth })));
          for (const img of imgs) {
            if (img.nw === 0) {
              pvFindings.push({ kind: 'broken-asset', route, viewport: vp.name, message: `img failed to load: ${img.src}` });
            }
          }

          const links = await page.$$eval('a[href]', (els) => els.map((e) => e.href));
          const linkChecks = await Promise.all(
            links.slice(0, 50).map(async (href) => {
              try {
                const r = await fetch(href, { method: 'HEAD' });
                return r.ok ? null : { href, status: r.status };
              } catch (err) {
                return { href, status: 0 };
              }
            }),
          );
          for (const lc of linkChecks) {
            if (lc) pvFindings.push({ kind: 'broken-asset', route, viewport: vp.name, message: `link ${lc.href} returned ${lc.status}` });
          }

          for (const m of consoleErrors) pvFindings.push({ kind: 'console', severity: 'error', route, viewport: vp.name, message: m });
          for (const m of consoleWarns) pvFindings.push({ kind: 'console', severity: 'warning', route, viewport: vp.name, message: m });
          for (const m of pageErrors) pvFindings.push({ kind: 'pageerror', route, viewport: vp.name, message: m });
        })(),
        timeoutMs,
        `scan ${route}@${vp.name}`,
      );
    } catch (err) {
      pvFindings.push({ kind: 'route-timeout', route, viewport: vp.name, message: err.message });
    }

    const outDir = path.join('artifacts', 'screenshots', route.replace(/[^a-z0-9]+/gi, '_'));
    fs.mkdirSync(outDir, { recursive: true });
    let screenshot = null;
    try {
      const buf = await page.screenshot({ fullPage: true });
      screenshot = path.join(outDir, `${vp.name}.png`);
      fs.writeFileSync(screenshot, buf);
    } catch (err) {
      pvFindings.push({ kind: 'screenshot-failed', route, viewport: vp.name, message: err.message });
    }

    await page.close();
    perViewport.push({ viewport: vp.name, screenshot, findings: pvFindings });
    findings.push(...pvFindings);
  }

  await context.close();
  return { route, perViewport, findings };
}
```

```javascript
import fs from 'node:fs';
import path from 'node:path';

function esc(s) {
  return String(s ?? '').replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}

export function buildHtml({ runId, generatedAt, grade, deltas, data, reportUrl }) {
  const rows = Object.entries(data.routes ?? {}).map(([route, r]) => {
    const a11y = r.findings.filter((f) => f.kind === 'a11y').length;
    const consoleErr = r.findings.filter((f) => f.kind === 'console' && f.severity === 'error').length;
    const perf = r.lighthouse?.performance ?? '-';
    const delta = (deltas.find((d) => d.route === route && d.metric === 'performance')?.delta) ?? 0;
    const deltaStr = delta === 0 ? '0' : (delta > 0 ? `+${delta}` : `${delta}`);
    return `<tr><td>${esc(route)}</td><td>${esc(a11y)}</td><td>${esc(consoleErr)}</td><td>${esc(perf)}</td><td>${esc(deltaStr)}</td></tr>`;
  }).join('');

  return `<!doctype html>
<html lang="en"><head><meta charset="utf-8"><title>UI/UX Audit Report — ${esc(runId)}</title>
<style>
body{font:14px/1.5 system-ui,sans-serif;max-width:960px;margin:24px auto;padding:0 16px;color:#222}
h1{margin-bottom:0}small{color:#666}
table{border-collapse:collapse;width:100%;margin:16px 0}
th,td{border:1px solid #ddd;padding:6px 10px;text-align:left}
th{background:#f4f4f4}
.crit{color:#b00020;font-weight:600}
.warn{color:#a05a00}
</style></head>
<body>
<h1>UI/UX Audit Report</h1>
<p><small>Run <code>${esc(runId)}</code> · generated ${esc(generatedAt)}${reportUrl ? ` · <a href="${esc(reportUrl)}">artifact</a>` : ''}</small></p>
<p><strong>Grade: ${esc(grade.grade)}</strong> · <span class="crit">${grade.critical.length} critical</span> · <span class="warn">${grade.warn.length} warn</span></p>
<table><thead><tr><th>Route</th><th>a11y</th><th>console errors</th><th>perf</th><th>Δ perf</th></tr></thead>
<tbody>${rows}</tbody></table>
</body></html>
`;
}

export function write({ runId, data, grade, deltas, outDir, reportUrl }) {
  fs.mkdirSync(outDir, { recursive: true });
  const jsonPath = path.join(outDir, 'report.json');
  const htmlPath = path.join(outDir, 'report.html');
  const json = { runId, generatedAt: new Date().toISOString(), grade, deltas, data };
  fs.writeFileSync(jsonPath, JSON.stringify(json, null, 2) + '\n', { encoding: 'utf8' });
  fs.writeFileSync(htmlPath, buildHtml({ runId, generatedAt: json.generatedAt, grade, deltas, data, reportUrl }), { encoding: 'utf8' });
  return { jsonPath, htmlPath };
}
```

```javascript
import fs from 'node:fs';

export function validate({ htmlPath }) {
  if (!fs.existsSync(htmlPath)) {
    throw new Error(`selfcheck: report missing at ${htmlPath}`);
  }
  const text = fs.readFileSync(htmlPath, 'utf8');
  if (text.length < 100) {
    throw new Error(`selfcheck: report suspiciously small (${text.length} bytes)`);
  }
  if (!text.includes('UI/UX Audit Report')) {
    throw new Error('selfcheck: report missing the "UI/UX Audit Report" header');
  }
  if (!/^<!doctype html>/i.test(text.trimStart())) {
    throw new Error('selfcheck: report does not start with a doctype');
  }
}
```

```javascript
export function buildBody({ runId, marker, data, deltas, grade, reportUrl }) {
  const routes = Object.entries(data.routes ?? {});
  const lines = [];
  lines.push(marker);
  lines.push('');
  lines.push(`## UI/UX Audit — run \`${runId}\``);
  lines.push('');
  lines.push(`**Grade: ${grade.grade}** · ${grade.critical.length} critical findings · ${grade.warn.length} warnings`);
  lines.push('');
  lines.push('| Route | A11y (axe) | Console | Perf (LH) | Δ vs base |');
  lines.push('|---|---|---|---|---|');
  for (const [route, r] of routes) {
    const a11y = r.findings?.filter((f) => f.kind === 'a11y').length ?? 0;
    const consoleErr = r.findings?.filter((f) => f.kind === 'console' && f.severity === 'error').length ?? 0;
    const perf = r.lighthouse?.performance ?? '-';
    const d = deltas.find((x) => x.route === route && x.metric === 'performance');
    const deltaStr = d ? (d.delta > 0 ? `+${d.delta}` : `${d.delta}`) : '0';
    const deltaCell = d && Math.abs(d.delta) >= 10 ? `**${deltaStr}**` : deltaStr;
    lines.push(`| \`${route}\` | ${a11y} | ${consoleErr} | ${perf} | ${deltaCell} |`);
  }
  lines.push('');
  if (grade.critical.length > 0) {
    lines.push('### Critical findings');
    lines.push('');
    for (const f of grade.critical) {
      lines.push(`- **${f.kind}** on \`${f.route}\`: ${f.message}`);
    }
    lines.push('');
  }
  if (reportUrl) {
    lines.push(`[Full HTML report](${reportUrl})`);
    lines.push('');
  }
  lines.push(`<sub>Run by \`@ui-audit-bot\`</sub>`);
  return lines.join('\n');
}

export async function postOrUpdate({ octokit, owner, repo, pr, body, marker }) {
  const { data: comments } = await octokit.rest.issues.listComments({ owner, repo, issue_number: pr, per_page: 100 });
  const existing = comments.find((c) => c.body?.includes(marker));
  if (existing) {
    const { data } = await octokit.rest.issues.updateComment({ owner, repo, comment_id: existing.id, body });
    return { id: data.id, action: 'updated' };
  }
  const { data } = await octokit.rest.issues.createComment({ owner, repo, issue_number: pr, body });
  return { id: data.id, action: 'created' };
}
```

```javascript
#!/usr/bin/env node
import { boot } from './lib/server.mjs';
import { login } from './lib/auth.mjs';
import { runRoute } from './lib/scan.mjs';
import { runOne as runLighthouse } from './lib/lighthouse.mjs';
import { ROUTES, VIEWPORTS } from './lib/routes.mjs';
import { read as readBaseline, write as writeBaseline, diff as diffBaseline, resolveRef } from './lib/baseline.mjs';
import { write as writeReport, buildHtml } from './lib/report.mjs';
import { compute as gradeCompute } from './lib/grade.mjs';
import { detect as detectCI, smokeLighthouse } from './lib/ci.mjs';
import { validate as selfcheck } from './lib/selfcheck.mjs';
import { postOrUpdate as postComment, buildBody } from './lib/comment.mjs';
import { Octokit } from '@octokit/rest';
import { chromium } from '@playwright/test';
import fs from 'node:fs';
import path from 'node:path';

function usage() {
  process.stdout.write(`usage: audit <run|baseline update|comment [run-id]>\n`);
}

async function cmdRun() {
  if (process.env.NODE_ENV === 'production') {
    throw new Error('audit refuses to run with NODE_ENV=production');
  }
  const ci = detectCI();
  if (ci) smokeLighthouse();

  const runId = new Date().toISOString().replace(/[:.]/g, '-');
  const outDir = path.join('artifacts', runId);
  fs.mkdirSync(outDir, { recursive: true });

  const port = 3000;
  const server = await boot({
    port,
    cwd: process.cwd(),
    command: ['pnpm', '--filter', '@finance-tracker/web', 'dev'],
    env: { PORT: String(port) },
  });

  const browser = await chromium.launch();
  let cookieHeader = null;

  try {
    if (process.env.TEST_USER_EMAIL && process.env.TEST_USER_PASSWORD) {
      try {
        const r = await login({
          baseUrl: server.url,
          email: process.env.TEST_USER_EMAIL,
          password: process.env.TEST_USER_PASSWORD,
        });
        cookieHeader = r.cookieHeader;
      } catch (err) {
        console.error(`auth: ${err.message}`);
      }
    }

    const data = { routes: {} };
    for (const route of ROUTES) {
      const r = await runRoute({ browser, baseUrl: server.url, route, viewports: VIEWPORTS, cookieHeader });
      const lh = await runLighthouse({ url: `${server.url}${route}`, outDir });
      data.routes[route] = { ...r, lighthouse: lh };
    }
  } finally {
    await browser.close();
    await server.stop();
  }

  const baseRef = ci?.baseRef ?? 'main';
  const { ref: compareRef } = resolveRef({ baseRef });
  const baseline = readBaseline(compareRef) ?? readBaseline('main');
  const deltas = diffBaseline(data, baseline);

  const findings = Object.values(data.routes).flatMap((r) => r.findings);
  const grade = gradeCompute({ findings, deltas, lighthouse: {} });

  const reportUrl = ci ? `https://github.com/${ci.owner}/${ci.repo}/actions/runs/${ci.runId}` : '';
  const { htmlPath, jsonPath } = writeReport({ runId, data, grade, deltas, outDir, reportUrl });
  selfcheck({ htmlPath });

  if (ci?.pr) {
    const octokit = new Octokit({ auth: ci.token });
    const marker = `<!-- ui-audit:run:${runId} -->`;
    const body = buildBody({ runId, marker, data, deltas, grade, reportUrl });
    await postOrUpdate({ octokit, owner: ci.owner, repo: ci.repo, pr: ci.pr, body, marker });
  }

  process.stdout.write(`grade: ${grade.grade}\n`);
  process.stdout.write(`report: ${htmlPath}\n`);

  if (grade.critical.length > 0) {
    process.stdout.write(`critical findings:\n`);
    for (const f of grade.critical) process.stdout.write(`  - ${f.kind} ${f.route}: ${f.message}\n`);
    process.exit(1);
  }
}

  if (process.env.NODE_ENV === 'production') {
    throw new Error('audit refuses to run with NODE_ENV=production');
  }
  smokeLighthouse();
  const port = 3000;
  const server = await boot({
    port,
    cwd: process.cwd(),
    command: ['pnpm', '--filter', '@finance-tracker/web', 'dev'],
    env: { PORT: String(port) },
  });

  const data = { routes: {} };
  try {
    if (process.env.TEST_USER_EMAIL && process.env.TEST_USER_PASSWORD) {
      const r = await login({ baseUrl: server.url, email: process.env.TEST_USER_EMAIL, password: process.env.TEST_USER_PASSWORD });
      cookieHeader = r.cookieHeader;

    for (const route of ROUTES) {
      const r = await runRoute({ browser, baseUrl: server.url, route, viewports: VIEWPORTS, cookieHeader });
      data.routes[route] = r;
    }
  } finally {
    await browser.close();
    await server.stop();
  }
  const summary = { version: 1, capturedAt: new Date().toISOString(), routes: {} };
  for (const [route, r] of Object.entries(data.routes)) {
    summary.routes[route] = {
      a11yViolations: r.findings.filter((f) => f.kind === 'a11y').length,
      consoleErrors: r.findings.filter((f) => f.kind === 'console' && f.severity === 'error').length,
    };
  }
  const { ref } = resolveRef({ baseRef: 'main' });
  const refKey = ref ?? 'main';
  const p = writeBaseline(refKey, summary);
  process.stdout.write(`baseline written: ${p}\n`);
}

async function cmdComment(_runId) {
  const ci = detectCI();
  if (!ci?.pr) {
    process.stdout.write('comment: not in a PR context, skipping\n');
    return;
  }
  const dir = path.join('artifacts', _runId ?? '');
  const jsonPath = path.join(dir, 'report.json');
  if (!fs.existsSync(jsonPath)) throw new Error(`report not found: ${jsonPath}`);
  const { runId, data, deltas, grade } = JSON.parse(fs.readFileSync(jsonPath, 'utf8').replace(/^\uFEFF/, ''));
  const reportUrl = `https://github.com/${ci.owner}/${ci.repo}/actions/runs/${ci.runId}`;
  const octokit = new Octokit({ auth: ci.token });
  const marker = `<!-- ui-audit:run:${runId} -->`;

  const r = await postOrUpdate({ octokit, owner: ci.owner, repo: ci.repo, pr: ci.pr, body, marker });
  process.stdout.write(`comment ${r.action}: ${r.id}\n`);
}

const [, , sub, ...rest] = process.argv;
if (!sub) { usage(); process.exit(2); }

try {
  if (sub === 'run') await cmdRun();
  else if (sub === 'baseline' && rest[0] === 'update') await cmdBaselineUpdate();
  else if (sub === 'comment') await cmdComment(rest[0]);
  else { usage(); process.exit(2); }
} catch (err) {
  process.stderr.write(`audit: ${err.message}\n`);
  process.exit(1);
}
```

```markdown
# @finance-tracker/ui-audit

UI/UX audit tool for `apps/web`. Runs Playwright + axe-core + Lighthouse against each route, diffs against a git-tracked baseline, posts a PR comment, and fails the build on critical findings.

## Quick start

\`\`\`
pnpm install
pnpm --filter @finance-tracker/ui-audit exec playwright install chromium
pnpm audit:ui
\`\`\`

Reports land in `tools/ui-audit/artifacts/<run-id>/`. The HTML report opens directly in a browser.

## Subcommands

- `pnpm audit:ui` — full audit against local dev server.
- `pnpm audit:ui:baseline` — capture the current state as a new baseline.
- `pnpm audit:ui:comment <run-id>` — re-post a comment for a past run.

## Environment

| Var | Required for | Notes |
|---|---|---|
| `TEST_USER_EMAIL` | Authenticated routes | Read by the test-only login endpoint |
| `TEST_USER_PASSWORD` | Authenticated routes | Same |
| `GITHUB_TOKEN` | PR comment | Provided by Actions automatically |
| `NODE_ENV` | Always | Tool refuses to run if `production` |

## Baselines

Baselines live in `tools/ui-audit/baselines/<ref>.json` and are committed to git. They are not large (a few KB each). To set the baseline for the current ref:

\`\`\`
pnpm audit:ui:baseline
git add tools/ui-audit/baselines/
git commit -m "chore(ui-audit): update baseline"
\`\`\`

## Failure policy

See `audit.config.example.yml` for the defaults. Strict by default: any P0 a11y, console error, broken asset, Lighthouse perf < 50, or perf regression > 10 → exit 1.
```

Create `tools/ui-audit/tests/smoke.mjs`:

```javascript
import { test } from 'node:test';
import assert from 'node:assert/strict';
import http from 'node:http';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { chromium } from '@playwright/test';
import { runRoute } from '../lib/scan.mjs';
import { write as writeReport } from '../lib/report.mjs';
import { compute as gradeCompute } from '../lib/grade.mjs';
import { validate as selfcheck } from '../lib/selfcheck.mjs';

let server;
let url;

test.before(async () => {
  server = http.createServer((req, res) => {
    if (req.url === '/' || req.url === '/smoke') {
      const html = fs.readFileSync(path.join(import.meta.dirname, 'smoke-page.html'), 'utf8');
      res.writeHead(200, { 'content-type': 'text/html' });
      res.end(html);
    } else {
      res.writeHead(404);
      res.end();
    }
  });
  await new Promise((r) => server.listen(0, '127.0.0.1', r));
  url = `http://127.0.0.1:${server.address().port}`;
});

test.after(() => server.close());

test('end-to-end produces a selfcheck-passing report', async () => {
  const browser = await chromium.launch();
  try {
    const r = await runRoute({
      browser,
      baseUrl: url,
      route: '/smoke',
      viewports: [{ name: 'desktop', width: 800, height: 600 }],
      timeoutMs: 15_000,
    });
    const findings = r.findings;
    const grade = gradeCompute({ findings, deltas: [], lighthouse: { performance: 99 } });
    const outDir = fs.mkdtempSync(path.join(os.tmpdir(), 'smoke-'));
    const runId = 'smoke-1';
    const { htmlPath } = writeReport({ runId, data: { routes: { '/smoke': r } }, grade, deltas: [], outDir });
    selfcheck({ htmlPath });
    assert.ok(fs.existsSync(htmlPath));
  } finally {
    await browser.close();
  }
});
```
